cmake_minimum_required(VERSION 3.16)
project(chat_db_tests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
get_filename_component(ROOT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
find_package(SQLite3 REQUIRED)
find_path(LOGGER_INCLUDE_DIR utils_log/logger.hpp
  HINTS "${ROOT_DIR}/build_dbg/_deps/utils_log-src" "${ROOT_DIR}/build_rel/_deps/utils_log-src"
  REQUIRED)
add_executable(chat_db_tests chat_db_tests.cpp ../src/chat_db.cpp "${ROOT_DIR}/src/sqlite3_helper.cpp")
target_include_directories(chat_db_tests PRIVATE ../src "${ROOT_DIR}/include" "${ROOT_DIR}/ui/shared" "${LOGGER_INCLUDE_DIR}")
if(TARGET SQLite3::SQLite3)
  target_link_libraries(chat_db_tests PRIVATE SQLite3::SQLite3)
else()
  target_link_libraries(chat_db_tests PRIVATE SQLite::SQLite3)
endif()
if(WIN32)
  target_link_libraries(chat_db_tests PRIVATE ole32 shell32)
endif()
add_test(NAME chat_db_tests COMMAND chat_db_tests)
